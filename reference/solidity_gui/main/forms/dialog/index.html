    SimpleFormDialog(form_class, form_renderer=default_form_renderer, initial_data=None, **kwargs) ¤

    Bases: QDialog

    A dedicated QDialog subclass for presenting a form to the user.

    This dialog displays the form according to the provided render function, adds an OK/Cancel button pair and wires up basic signal handlers. In the event of the submitted form failing validation, the dialog contents are re-rendered with error messages.

    This class is designed to be extensible and indeed must be extended to in order to do anything useful. The _ok_clicked handler provides sensible default behaviour in the case of invalid data being provided but does not do anything in the case of valid data.

    The simplest intended use case for this form is therefore as below:

    class FunctionalDialog(SimpleFormDialog):
    def _ok_clicked(self):
        if data := super()._ok_clicked():
            # do something useful with data

    This preserves the data validation behaviour and tacks on additional actions to perform with data that passes validation. Alternatively, you can overwrite _ok_clicked without calling the superclass implementation. This gives more flexibility but you will probably need to implement data validation and form update behaviours.

    A basic StyleSheet is applied that assumes the naming conventions used by default_form_renderer. The headers of fields in top-level forms are shown in bold and without bolding for nested fields.

    def __init__(
    self,
    form_class: type[Form],
    form_renderer: Callable[[FormInfo], QLayout] = default_form_renderer,
    initial_data: dict[str, Any] | None = None,
    **kwargs: Any,
):
    """Initialise a new SimpleFormDialog.

    Args:
      form_class: form to display
      form_renderer: renderer that creates GUI elements from a FormInfo instance
      initial_data: data to pre-populate into the form
    """
    super().__init__(**kwargs)
    self.form_class = form_class
    self.form_info = self.form_class().form_info(data=initial_data or dict())
    self.form_renderer = form_renderer
    contents = self._create_contents(self.form_info)
    self.setLayout(contents)
    # style different label types in form
    self.setStyleSheet(
        "QLabel#error { color: red } "
        "QLabel#description { color: gray } "
        "QLabel#header-0 { font-weight: bold }"
    )
    self._data: None | dict[str, Any] = None

    def default_field_renderer(
    field_info: FieldInfo,
    level: int,
    include_header: bool = True,
) -> QLayout:
    """Render an individual field within a form.

    Args:
      field_info: the field to render
      level: the level of nesting for the form containing this field
      include_header: whether to render field_info.header in a QLabel
    """
    layout = QVBoxLayout()
    if include_header and field_info.header:
        layout.addWidget(
            _create_named_label(f"{field_info.header.capitalize()}:", f"header-{level}")
        )
    if field_info.description:
        layout.addWidget(_create_named_label(field_info.description, "description"))

    for i, input_ in enumerate(field_info.input):
        if isinstance(field_info.errors, dict):
            if errors := field_info.errors.get(i, None):
                for error in errors:
                    layout.addWidget(_create_named_label(error, "error"))
        layout.addWidget(input_)
    if isinstance(field_info.errors, list):
        for error in field_info.errors:
            layout.addWidget(_create_named_label(error, "error"))
    return layout

    default_form_renderer(form_info, level=0) ¤

    Render a FormInfo instance into a QVBoxLayout.

    This function aims to provide a sensible default layout agnostic to use context. Field errors, headers and descriptions are displayed as QLabel objects with appropriate ObjectName values to allow simple and consistent styling. Errors are displayed immediately after the most relevant context (e.g. input widget for field level errors, end of form for schema level errors).

    Rendering of nested forms is supported. The level of nesting is tracked via level and is used to set different ObjectNames for field headers i.e. the QLabel's used to display header information at the outer level have the ObjectName 'header-0', at the first level of nesting 'header-1' and so on. This can be used in combination with the StyleSheet of a dialog to control the appearance of headers at different levels. Nested forms are also rendered with a small indent on the left.

    def default_form_renderer(form_info: FormInfo, level: int = 0) -> QLayout:
    """Render a FormInfo instance into a QVBoxLayout.

    This function aims to provide a sensible default layout agnostic to use
    context. Field errors, headers and descriptions are displayed as QLabel objects with
    appropriate ObjectName values to allow simple and consistent styling. Errors are
    displayed immediately after the most relevant context (e.g. input widget for field
    level errors, end of form for schema level errors).

    Rendering of nested forms is supported. The level of nesting is tracked via `level`
    and is used to set different ObjectNames for field headers i.e. the QLabel's used to
    display header information at the outer level have the ObjectName 'header-0', at the
    first level of nesting 'header-1' and so on. This can be used in combination with
    the StyleSheet of a dialog to control the appearance of headers at different
    levels. Nested forms are also rendered with a small indent on the left.

    Args:
      form_info: description of the form to render
      level: the level of nesting of the form to render used to fine tune appearance
    """
    layout = QVBoxLayout()

    for info in form_info.fields.values():
        if isinstance(info, FieldInfo):
            layout.addLayout(default_field_renderer(info, level))
        elif isinstance(info, FormInfo):
            if info.header:
                layout.addWidget(
                    _create_named_label(
                        f"{info.header.capitalize()}:", f"header-{level}"
                    )
                )
            if info.single_choice:
                nested_layout = single_choice_form_renderer(info, level + 1)
            else:
                nested_layout = default_form_renderer(info, level + 1)
                # add to left margin to indent nested forms a bit
                margins = nested_layout.getContentsMargins()
                nested_layout.setContentsMargins(margins[0] + 20, *margins[1:])
            layout.addLayout(nested_layout)

    for error in form_info.schema_errors:
        layout.addWidget(_create_named_label(error, "error"))
    return layout

    def single_choice_field_renderer(
    field_info: FieldInfo, level: int, checked: bool = False
) -> QLayout:
    """Return layout of a form field with an associated radio button.

    Rather than using a QLabel for the header a radio button is used instead. The
    disabled attribute of the field_info object is set according to the checked status
    of the radio button.
    """
    layout = QVBoxLayout()
    radio = QRadioButton(field_info.header)

    field_info.enabled = checked
    field_layout = default_field_renderer(field_info, level, include_header=False)

    # set up signal handler that controls whether field input widgets are disabled
    def _toggled_handler(checked: bool) -> None:
        field_info.enabled = checked

    radio.toggled.connect(_toggled_handler)  # type: ignore[attr-defined]

    radio.setChecked(checked)
    layout.addWidget(radio)
    layout.addLayout(field_layout)
    return layout

    def single_choice_form_renderer(form_info: FormInfo, level: int = 0) -> QLayout:
    """Render a FormInfo instance such that only a single field may be completed.

    This is used to present the user with a choice between data to provide. Practically
    this is achieved via the use of a QGroupBox widget with exclusive radio buttons for
    each field in `form_info`.

    The presence of nested forms is not currently supported.
    """
    layout = QVBoxLayout()
    group_box = QGroupBox()
    layout.addWidget(group_box)
    box_layout = QVBoxLayout()
    group_box.setLayout(box_layout)

    for i, (name, field_info) in enumerate(form_info.fields.items()):
        if isinstance(field_info, FormInfo):
            raise ValueError(
                "Nesting forms within a SingleChoiceNestedForm is not supported."
            )
        box_layout.addLayout(
            single_choice_field_renderer(field_info, level, checked=not i)
        )
    for error in form_info.schema_errors:
        layout.addWidget(_create_named_label(error, "error"))
    return layout
